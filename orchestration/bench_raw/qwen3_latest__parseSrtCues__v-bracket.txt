function parseSrtCues(srt: string): { start: number; end: number; text: string }[] {
  // Normalize line endings
  let normalized = srt.replace(/\r/g, '');
  // Split into lines
  const lines = normalized.split('\n');
  // Group into blocks
  const blocks: string[][] = [];
  let currentBlock: string[] = [];
  for (const line of lines) {
    if (line.trim() === '') {
      if (currentBlock.length > 0) {
        blocks.push(currentBlock);
        currentBlock = [];
      }
    } else {
      currentBlock.push(line);
    }
  }
  if (currentBlock.length > 0) {
    blocks.push(currentBlock);
  }

  const result: { start: number; end: number; text: string }[] = [];

  for (const block of blocks) {
    // Each block should have at least two lines: index and time
    if (block.length < 2) continue;
    const timeLine = block[1];
    // Check if the time line matches the pattern
    const timeMatch = timeLine.match(/^(.*?),.*? --> .*?(,.*)$/);
    if (!timeMatch) continue;
    const [startStr, endStr] = timeMatch.slice(1, 3);
    // Parse start and end times
    const startSec = parseTimeToSeconds(startStr);
    const endSec = parseTimeToSeconds(endStr);
    if (startSec === null || endSec === null) continue;
    // Collect text lines: all lines after the time line
    const textLines = block.slice(2).map(line => line.trim());
    const text = textLines.join('\n');
    result.push({ start: startSec, end: endSec, text });
  }

  return result;
}

function parseTimeToSeconds(timeStr: string): number | null {
  // timeStr is like "HH:MM:SS,mmm"
  const parts = timeStr.split(',');
  if (parts.length !== 2) return null;
  const [timePart, milli] = parts;
  const [h, m, s] = timePart.split(':').map(Number);
  if (isNaN(h) || isNaN(m) || isNaN(s)) return null;
  let totalSec = h * 3600 + m * 60 + s;
  const ms = Number(milli);
  if (!isNaN(ms)) {
    totalSec += ms / 1000;
  }
  return totalSec;
}
