export function sumClipDurations(clips: { start: number; end: number }[]): number {
    let total = 0;
    for (const clip of clips) {
        const start = clip.start;
        const end = clip.end;
        if (
            !Number.isFinite(start) ||
            !Number.isFinite(end) ||
            end <= start
        ) {
            continue;
        }
        total += end - start;
    }
    return Math.round(total * 1000) / 1000;
}
