export function formatTimecode(sec: number): string {
  if (!isFinite(sec) || sec < 0) return "00:00.000";
  
  const totalMs = Math.round(sec * 1000);
  let minutes = Math.floor(totalMs / (60 * 1000));
  let remainingMs = totalMs % (60 * 1000);
  let seconds = Math.floor(remainingMs / 1000);
  remainingMs %= 1000;
  
  return (
    String.padStart(minutes, 2, "0") +
    ":" +
    String.padStart(seconds, 2, "0") +
    "." +
    String padStart(remainingMs, 3, "0")
  );
}
