    ```typescript
    export function sumClipDurations(clips: { start: number; end: number }[]): number {
      let total = 0;
      for (const clip of clips) {
        const { start, end } = clip;
        if (Number.isFinite(start) && Number.isFinite(end) && end > start) {
          total += (end - start);
        }
      }
    ```

    ```typescript
export function sumClipDurations(clips: { start: number; end: number }[]): number {
    ```

      Number.isFinite(clip.end) &&
      clip.end > clip.start
    ) {

    }
  }
