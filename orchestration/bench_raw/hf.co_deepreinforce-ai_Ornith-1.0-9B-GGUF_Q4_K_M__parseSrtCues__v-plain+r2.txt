    export function parseSrtCues(srt: string): { start: number; end: number; text: string }[] {
      if (!srt || srt.trim() === "") return [];

      const lines = srt.split(/\r?\n/).map(l => l.trim());
      const result: { start: number; end: number; text: string }[] = [];

      let blockLines: string[] = [];

      for (const line of lines) {
        if (!line) {
          if (blockLines.length > 0) {
            processBlock(blockLines, result);
            blockLines = [];
          }
        } else {
          blockLines.push(line);
        }
      }

      // Handle trailing block without final newline? SRT usually ends with newline. But just in case.
      if (blockLines.length > 0) {
        processBlock(blockLines, result);
      }

      return result;
    }

    function processBlock(lines: string[], result: { start: number; end: number; text: string }[]) {
      // Find the time line. In standard SRT it's index \n time \n text...
      // But we just need to find the valid time line within the block.
      const timeLineIndex = lines.findIndex(l => /^(\d{2}):(\d{2}):(\d{2}),(\d{3})\s*-->\s*(\d{2}):(\d{2}):(\d{2}),(\d{3})$/.test(l));

      if (timeLineIndex === -1) return; // Malformed

      const [startStr, endStr] = lines[timeLineIndex].split("-->");
      const startSec = parseTime(startStr.trim());
      const endSec = parseTime(endStr.trim());

      const textLines = lines.slice(timeLineIndex + 1);
      result.push({
        start: startSec,
        end: endSec,
        text: textLines.join("\n")
      });
    }

    function parseTime(t: string): number {
      // HH:MM:SS,mmm
      const parts = t.split(",").map(Number);
      if (parts.length !== 2) return 0;
      const [hms, ms] = parts;
      const [h, m, s] = hms.split(":").map(Number);
      return h * 3600 + m * 60 + s + ms / 1000;
    }
